// Prime Option backend v2 — Daraja STK + trader accounts (PIN) + server-side ledger
// Env vars (set in Render dashboard, never commit secrets):
//   DARAJA_CONSUMER_KEY, DARAJA_CONSUMER_SECRET,
//   DARAJA_SHORTCODE (default 174379), DARAJA_PASSKEY,
//   DARAJA_ENV = sandbox | production   (default sandbox)
//   ADMIN_KEY (strong secret for admin endpoints — REQUIRED, set in Render)
//   B2C_INITIATOR, B2C_SECURITY_CREDENTIAL, B2C_SHORTCODE (auto payouts, production)
const express = require('express');
const axios = require('axios');
const cors = require('cors');
const crypto = require('crypto');
const fs = require('fs');
const path = require('path');
const app = express();
app.use(cors());
app.use(express.json());

// ---------- config ----------
const CONSUMER_KEY = process.env.DARAJA_CONSUMER_KEY || '';
const CONSUMER_SECRET = process.env.DARAJA_CONSUMER_SECRET || '';
const SHORTCODE = process.env.DARAJA_SHORTCODE || '174379';
const PASSKEY = process.env.DARAJA_PASSKEY || 'bfb279f9aa9bdbcf158e97dd71a467cd2e0c893059b10f78e6b72ada1ed2c919';
const IS_SANDBOX = (process.env.DARAJA_ENV || 'sandbox').toLowerCase() !== 'production';
const BASE_URL = IS_SANDBOX ? 'https://sandbox.safaricom.co.ke' : 'https://api.safaricom.co.ke';
const ADMIN_KEY = process.env.ADMIN_KEY || '';
const B2C_INITIATOR = process.env.B2C_INITIATOR || '';
const B2C_SEC_CRED = process.env.B2C_SECURITY_CREDENTIAL || '';
const B2C_SHORTCODE = process.env.B2C_SHORTCODE || SHORTCODE;
if (!ADMIN_KEY) console.warn('WARNING: set a strong ADMIN_KEY env var!');

// ---------- tiny JSON db ----------
// NOTE: Render free tier filesystem is ephemeral — db.json resets on redeploy.
// Fine for testing. For real-money production attach a Render Disk or use Postgres.
const DB_PATH = path.join(__dirname, 'data', 'db.json');
function loadDb() {
  try { return JSON.parse(fs.readFileSync(DB_PATH, 'utf8')); }
  catch (e) { return { users: {}, txns: [], withdrawals: [] }; }
}
function saveDb(db) {
  fs.mkdirSync(path.dirname(DB_PATH), { recursive: true });
  fs.writeFileSync(DB_PATH, JSON.stringify(db, null, 2));
}

// ---------- auth helpers ----------
const sessions = new Map(); // token -> phone
function normPhone(p) {
  let d = String(p || '').replace(/\D/g, '');
  if (/^0[17]\d{8}$/.test(d)) d = '254' + d.slice(1);
  if (!/^254[17]\d{8}$/.test(d)) throw new Error('Invalid Safaricom number. Use 07... or 2547...');
  return d;
}
function hashPin(pin, salt) { return crypto.scryptSync(String(pin), salt, 32).toString('hex'); }
function authMw(req, res, next) {
  const t = (req.headers.authorization || '').replace(/^Bearer\s+/, '');
  const phone = sessions.get(t);
  if (!phone) return res.status(401).json({ success: false, error: 'Not logged in' });
  req.phone = phone; next();
}
function adminMw(req, res, next) {
  if (!ADMIN_KEY || req.headers['x-admin-key'] !== ADMIN_KEY)
    return res.status(403).json({ success: false, error: 'Forbidden' });
  next();
}

// ---------- daraja ----------
let tokenCache = { token: null, expiresAt: 0 };
async function getAccessToken() {
  if (Date.now() < tokenCache.expiresAt && tokenCache.token) return tokenCache.token;
  if (!CONSUMER_KEY || !CONSUMER_SECRET) throw new Error('DARAJA_CONSUMER_KEY / DARAJA_CONSUMER_SECRET not set');
  const auth = Buffer.from(`${CONSUMER_KEY}:${CONSUMER_SECRET}`).toString('base64');
  const res = await axios.get(`${BASE_URL}/oauth/v1/generate?grant_type=client_credentials`, {
    headers: { Authorization: `Basic ${auth}` }, timeout: 15000
  });
  tokenCache = { token: res.data.access_token, expiresAt: Date.now() + 3000 * 1000 };
  return tokenCache.token;
}
function timestamp() {
  const d = new Date(), p = (n, l = 2) => String(n).padStart(l, '0');
  return `${d.getFullYear()}${p(d.getMonth()+1)}${p(d.getDate())}${p(d.getHours())}${p(d.getMinutes())}${p(d.getSeconds())}`;
}
const stkTxns = new Map(); // CheckoutRequestID -> {status, phone, amount, userPhone, receipt, resultDesc}

// ---------- public ----------
app.get('/', (req, res) => res.send(`Prime Option backend v2 OK — ${IS_SANDBOX ? 'SANDBOX' : 'PRODUCTION'}`));
app.get('/health', (req, res) => {
  const db = loadDb();
  res.json({ ok: true, version: 2, mode: IS_SANDBOX ? 'sandbox' : 'production',
             shortcode: SHORTCODE, users: Object.keys(db.users).length });
});
app.get('/api/test-auth', async (req, res) => {
  try { const t = await getAccessToken();
    res.json({ success: true, message: 'Daraja credentials VALID', tokenPreview: t.slice(0, 12) + '...' });
  } catch (e) { res.status(500).json({ success: false, error: e.response?.data || e.message }); }
});

// ---------- trader accounts (phone + PIN) ----------
app.post('/api/auth/register', (req, res) => {
  try {
    const phone = normPhone(req.body.phone);
    const pin = String(req.body.pin || '');
    if (!/^\d{4,6}$/.test(pin)) throw new Error('PIN must be 4-6 digits');
    const db = loadDb();
    if (db.users[phone]) return res.status(400).json({ success: false, error: 'Account exists — log in instead' });
    const salt = crypto.randomBytes(16).toString('hex');
    db.users[phone] = { pinHash: hashPin(pin, salt), salt, balance: 0, createdAt: Date.now() };
    saveDb(db);
    const token = crypto.randomBytes(24).toString('hex');
    sessions.set(token, phone);
    res.json({ success: true, token, phone, balance: 0 });
  } catch (e) { res.status(400).json({ success: false, error: e.message }); }
});
app.post('/api/auth/login', (req, res) => {
  try {
    const phone = normPhone(req.body.phone);
    const pin = String(req.body.pin || '');
    const db = loadDb();
    const u = db.users[phone];
    if (!u || hashPin(pin, u.salt) !== u.pinHash) throw new Error('Wrong phone or PIN');
    const token = crypto.randomBytes(24).toString('hex');
    sessions.set(token, phone);
    res.json({ success: true, token, phone, balance: u.balance });
  } catch (e) { res.status(401).json({ success: false, error: e.message }); }
});
app.get('/api/account', authMw, (req, res) => {
  const db = loadDb();
  const u = db.users[req.phone];
  res.json({ success: true, phone: req.phone, balance: u ? u.balance : 0 });
});

// ---------- STK deposit (auto-credits server balance on callback) ----------
app.post('/api/deposit/stkpush', async (req, res) => {
  try {
    const t = (req.headers.authorization || '').replace(/^Bearer\s+/, '');
    const authedPhone = sessions.get(t);
    const phone = authedPhone || normPhone(req.body.phone);
    const amount = Math.round(Number(req.body.amount));
    if (!amount || amount < 1) throw new Error('Amount must be at least 1');
    const token = await getAccessToken();
    const ts = timestamp();
    const password = Buffer.from(`${SHORTCODE}${PASSKEY}${ts}`).toString('base64');
    const cbUrl = `${req.protocol}://${req.get('host')}/api/mpesa/callback`;
    const r = await axios.post(`${BASE_URL}/mpesa/stkpush/v1/processrequest`, {
      BusinessShortCode: SHORTCODE, Password: password, Timestamp: ts,
      TransactionType: 'CustomerPayBillOnline', Amount: amount,
      PartyA: phone, PartyB: SHORTCODE, PhoneNumber: phone,
      CallBackURL: cbUrl, AccountReference: 'PRIMEOPTION', TransactionDesc: 'Prime Option deposit'
    }, { headers: { Authorization: `Bearer ${token}` }, timeout: 20000 });
    const crid = r.data.CheckoutRequestID;
    stkTxns.set(crid, { status: 'PENDING', phone, amount, userPhone: authedPhone || null, createdAt: Date.now() });
    res.json({ success: true, checkoutRequestId: crid, message: 'STK push sent — enter M-Pesa PIN on ' + phone });
  } catch (e) { res.status(500).json({ success: false, error: e.response?.data || e.message }); }
});
app.post('/api/mpesa/callback', (req, res) => {
  try {
    const cb = req.body?.Body?.stkCallback;
    if (cb) {
      const t = stkTxns.get(cb.CheckoutRequestID);
      if (t) {
        const db = loadDb();
        if (cb.ResultCode == 0) {
          const items = cb.CallbackMetadata?.Item || [];
          const get = (n) => (items.find(i => i.Name === n) || {}).Value;
          t.status = 'SUCCESS'; t.receipt = get('MpesaReceiptNumber'); t.resultDesc = cb.ResultDesc;
          if (t.userPhone && db.users[t.userPhone]) {           // AUTO-CREDIT
            db.users[t.userPhone].balance += t.amount;
            db.txns.push({ id: 'dep_' + Date.now(), type: 'deposit', phone: t.userPhone,
                           amount: t.amount, receipt: t.receipt, status: 'SUCCESS', createdAt: Date.now() });
            saveDb(db);
          }
        } else { t.status = 'FAILED'; t.resultDesc = cb.ResultDesc; }
      }
    }
  } catch (e) { console.error('callback err', e.message); }
  res.json({ ResultCode: 0, ResultDesc: 'Accepted' });
});
app.get('/api/deposit/status/:crid', (req, res) => {
  const t = stkTxns.get(req.params.crid);
  if (!t) return res.status(404).json({ success: false, error: 'Unknown transaction' });
  res.json({ success: true, ...t });
});

// ---------- withdrawals ----------
app.post('/api/withdraw/request', authMw, (req, res) => {
  try {
    const amount = Math.round(Number(req.body.amount));
    if (!amount || amount < 10) throw new Error('Minimum withdrawal is 10');
    const db = loadDb();
    const u = db.users[req.phone];
    if (!u || u.balance < amount) throw new Error('Insufficient balance');
    u.balance -= amount;
    const w = { id: 'wd_' + Date.now(), phone: req.phone, amount, status: 'PENDING', createdAt: Date.now() };
    db.withdrawals.push(w);
    db.txns.push({ id: w.id, type: 'withdraw', phone: req.phone, amount, status: 'PENDING', createdAt: Date.now() });
    saveDb(db);
    res.json({ success: true, withdrawal: w, balance: u.balance });
  } catch (e) { res.status(400).json({ success: false, error: e.message }); }
});
app.get('/api/withdrawals', authMw, (req, res) => {
  const db = loadDb();
  res.json({ success: true, withdrawals: db.withdrawals.filter(w => w.phone === req.phone) });
});

// ---------- admin (x-admin-key header) ----------
app.get('/api/admin/users', adminMw, (req, res) => {
  const db = loadDb();
  res.json({ success: true, users: Object.entries(db.users).map(([phone, u]) =>
    ({ phone, balance: u.balance, createdAt: u.createdAt })) });
});
app.get('/api/admin/transactions', adminMw, (req, res) => {
  const db = loadDb();
  res.json({ success: true, txns: db.txns.slice(-200).reverse(), withdrawals: db.withdrawals.slice().reverse() });
});
app.post('/api/admin/adjust', adminMw, (req, res) => {
  try {
    const phone = normPhone(req.body.phone);
    const amount = Math.round(Number(req.body.amount));
    if (!amount) throw new Error('Amount required (negative to debit)');
    const db = loadDb();
    if (!db.users[phone]) throw new Error('No such user');
    db.users[phone].balance += amount;
    db.txns.push({ id: 'adj_' + Date.now(), type: 'adjust', phone, amount,
                   reason: req.body.reason || '', status: 'SUCCESS', createdAt: Date.now() });
    saveDb(db);
    res.json({ success: true, balance: db.users[phone].balance });
  } catch (e) { res.status(400).json({ success: false, error: e.message }); }
});
// Automated M-Pesa payout (B2C) — needs B2C_INITIATOR + B2C_SECURITY_CREDENTIAL, production only
app.post('/api/admin/payout', adminMw, async (req, res) => {
  try {
    if (!B2C_INITIATOR || !B2C_SEC_CRED) throw new Error('B2C not configured — set B2C_INITIATOR and B2C_SECURITY_CREDENTIAL');
    const phone = normPhone(req.body.phone);
    const amount = Math.round(Number(req.body.amount));
    if (!amount || amount < 10) throw new Error('Minimum payout is 10');
    const token = await getAccessToken();
    const r = await axios.post(`${BASE_URL}/mpesa/b2c/v1/paymentrequest`, {
      InitiatorName: B2C_INITIATOR, SecurityCredential: B2C_SEC_CRED,
      CommandID: 'BusinessPayment', Amount: amount, PartyA: B2C_SHORTCODE, PartyB: phone,
      Remarks: 'Prime Option withdrawal',
      QueueTimeOutURL: `${req.protocol}://${req.get('host')}/api/mpesa/b2c-timeout`,
      ResultURL: `${req.protocol}://${req.get('host')}/api/mpesa/b2c-result`,
      Occasion: 'withdrawal'
    }, { headers: { Authorization: `Bearer ${token}` }, timeout: 20000 });
    const db = loadDb();
    db.txns.push({ id: 'b2c_' + Date.now(), type: 'payout', phone, amount, status: 'SENT', createdAt: Date.now() });
    saveDb(db);
    res.json({ success: true, result: r.data });
  } catch (e) { res.status(500).json({ success: false, error: e.response?.data || e.message }); }
});
app.post('/api/mpesa/b2c-result', (req, res) => { res.json({ ResultCode: 0, ResultDesc: 'ok' }); });
app.post('/api/mpesa/b2c-timeout', (req, res) => { res.json({ ResultCode: 0, ResultDesc: 'ok' }); });

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => console.log(`Prime backend v2 on :${PORT} (${IS_SANDBOX ? 'sandbox' : 'production'})`));
